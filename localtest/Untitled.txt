apiVersion: apps/v1
kind: Deployment
metadata:
  name: deployprod
  namespace: newprod
spec:
  replicas: 2
  selector:
    matchLabels:
      app: server
  template:
    metadata:
      labels:
        app: server
    spec:
      volumes:
      - name: newvol1
        emptyDir: {}
      - name: newconfig
        configMap:
          name: nginx-html-config
      initContainers:
      - name: init-copy-html
        image: busybox
        command: ["sh", "-c", "cp /config/index.html /html/index.html"]
        volumeMounts:
        - name: newconfig
          mountPath: /config
        - name: newvol1
          mountPath: /html
      containers:
      - name: c01
        image: nginx:latest
        ports:
        - containerPort: 80
        env:
        - name: ENVIRONMENT
          value: PROD
        volumeMounts:
        - name: newvol1
          mountPath: /usr/share/nginx/html
      - name: c02
        image: ubuntu:latest
        command: ["/bin/bash", "-c", "sleep 1000"]
        env:
        - name: ENVIRONMENT
          value: DEV
        volumeMounts:
        - name: newvol1
          mountPath: /local/test
---
